const FY_REGEX = /^\d{4}-\d{2,4}$/;

function onOpen() {
    const ui = SpreadsheetApp.getUi();
    ui.createMenu('Financial Dashboard')
        .addItem('Add New Financial Year', 'addNewYearPrompt')
        .addItem('Update Master Year & Dashboard', 'updateAllSheetsManual')
        .addSeparator()
        .addItem('Export to PDF (All Sheets or by Name)', 'exportPdfPrompt')
        .addToUi();
}

function addNewYearPrompt() {
    const ui = SpreadsheetApp.getUi();
    const result = ui.prompt(
        'Add New Financial Year Sheet',
        'Enter the financial year range (e.g., 2025-2026 or 2025-26):',
        ui.ButtonSet.OK_CANCEL
    );

    const button = result.getSelectedButton();
    const fyText = result.getResponseText().trim();

    if (button !== ui.Button.OK) return;

    const match = fyText.match(/^(\d{4})-(\d{2}|\d{4})$/);
    if (!match) {
        ui.alert('Invalid Format', 'Please enter a valid financial year range like "2025-2026" or "2025-26".', ui.ButtonSet.OK);
        return;
    }

    let startYear = parseInt(match[1], 10);
    let endYear = match[2].length === 2
        ? Math.floor(startYear / 100) * 100 + parseInt(match[2], 10)
        : parseInt(match[2], 10);

    if (endYear !== startYear + 1) {
        ui.alert('Invalid Year Range', 'The second year must be exactly one year after the first year (e.g., 2025-2026).', ui.ButtonSet.OK);
        return;
    }

    const normalizedFY = `${startYear}-${endYear}`;
    addNewYearSheet(normalizedFY, startYear, endYear);
}

function addNewYearSheet(fyRange, startYear, endYear) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (ss.getSheetByName(fyRange)) {
        SpreadsheetApp.getUi().alert(`A sheet named "${fyRange}" already exists!`);
        return;
    }

    if (!startYear || !endYear) {
        const parts = fyRange.split("-");
        startYear = parseInt(parts[0], 10);
        endYear = parts[1].length === 2 ? Math.floor(startYear / 100) * 100 + parseInt(parts[1], 10) : parseInt(parts[1], 10);
    }

    const newSheet = ss.insertSheet(fyRange);

    setupTableHeaders(newSheet);

    const months = [
        [`July, ${startYear}`], [`August, ${startYear}`], [`September, ${startYear}`],
        [`October, ${startYear}`], [`November, ${startYear}`], [`December, ${startYear}`],
        [`January, ${endYear}`], [`February, ${endYear}`], [`March, ${endYear}`],
        [`April, ${endYear}`], [`May, ${endYear}`], [`June, ${endYear}`]
    ];
    newSheet.getRange("A4:A15").setValues(months).setNumberFormat("@");

    // Batch set formulas across ranges (auto-adjusts row references)
    newSheet.getRange("F4:F15").setFormula("=SUM(B4:E4)");
    newSheet.getRange("H4:H15").setFormula("=F4-N(G4)-N(J4)");

    // Batch set alternating row backgrounds
    const bgs = [];
    for (let r = 4; r <= 15; r++) {
        const bg = (r % 2 === 0) ? "#FFFFFF" : "#F8F9FA";
        bgs.push(Array(11).fill(bg));
    }
    newSheet.getRange("A4:K15").setBackgrounds(bgs);

    // Total Row
    newSheet.getRange("A16").setValue(`Total (${fyRange})`);
    const sumFormulas = [
        "=SUM(B4:B15)", "=SUM(C4:C15)", "=SUM(D4:D15)", "=SUM(E4:E15)",
        "=SUM(F4:F15)", "=SUM(G4:G15)", "=SUM(H4:H15)", "",
        "=SUM(J4:J15)", "=SUM(K4:K15)"
    ];
    newSheet.getRange("B16:K16").setFormulas([sumFormulas]);

    newSheet.getRange("A16:K16").setFontWeight("bold").setBackground("#D5F5E3");

    newSheet.getRange("B4:H16").setNumberFormat('#,##0"/="');
    newSheet.getRange("I4:I16").setNumberFormat("@");
    newSheet.getRange("J4:K16").setNumberFormat('#,##0"/="');

    applyProfessionalBorders(newSheet.getRange("A2:K16"));
    newSheet.autoResizeColumns(1, 11);
    trimSheet(newSheet, 16, 11);

    updateAllSheets(newSheet, ss);
    SpreadsheetApp.getUi().alert('Success', `Sheet for Financial Year ${fyRange} created!`, SpreadsheetApp.getUi().ButtonSet.OK);
}

function setupTableHeaders(sheet) {
    sheet.getRange("A1:K3").breakApart();

    sheet.getRange("A2:A3").merge().setValue("Month/Year");
    sheet.getRange("I2:I3").merge().setValue("Remarks");
    sheet.getRange("J2:J3").merge().setValue("Provident Fund");
    sheet.getRange("K2:K3").merge().setValue("Monthly Savings");

    sheet.getRange("B2:H2").merge().setValue("Salary and Others");

    const subHeaders = [[
        "Basic Salary", "House Rent", "Conveyance Allowance",
        "Medical & Others", "Total Salary", "Tax", "Net Income"
    ]];
    sheet.getRange("B3:H3").setValues(subHeaders);

    sheet.getRange("A2:K3")
        .setFontWeight("bold")
        .setFontColor("#FFFFFF")
        .setBackground("#2C3E50")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    sheet.getRange("B3:H3")
        .setFontColor("#1C2833")
        .setBackground("#EAEDED");
}

function cleanCurrencyInput(range, customFormat = null) {
    const values = range.getValues();
    let hasChanges = false;

    for (let i = 0; i < values.length; i++) {
        for (let j = 0; j < values[i].length; j++) {
            const val = values[i][j];
            if (typeof val === 'string' && val.trim().length > 0) {
                const cleaned = val.replace(/,/g, '').replace(/[^0-9.]/g, '');
                if (cleaned !== '' && !isNaN(cleaned)) {
                    values[i][j] = parseFloat(cleaned);
                    hasChanges = true;
                }
            }
        }
    }

    if (hasChanges) {
        range.setValues(values);
        if (customFormat) range.setNumberFormat(customFormat);
    }
}

function onEdit(e) {
    if (!e || !e.range) return;

    const sheet = e.range.getSheet();
    const sheetName = sheet.getName();

    // 1. Handlers for Expenses, DPS Tracker & Cash in Hand Sheets
    if (sheetName === "Expenses" || sheetName === "DPS Tracker" || sheetName === "Cash in Hand") {
        const startCol = e.range.getColumn();
        const endCol = startCol + e.range.getNumColumns() - 1;
        const rowIndex = e.range.getRow();

        // Clean currency strings if entered in Amount column (Col 3, or Col 2 for Cash in Hand) or Initial Amount (Col 2 in Expenses)
        const isExpenseAmt = (sheetName === "Expenses" && startCol <= 3 && endCol >= 3);
        const isExpenseInit = (sheetName === "Expenses" && startCol <= 2 && endCol >= 2 && rowIndex >= 5 && rowIndex <= 7);
        const isDpsAmt = (sheetName === "DPS Tracker" && startCol <= 3 && endCol >= 3);
        const isCashAmt = (sheetName === "Cash in Hand" && startCol <= 2 && endCol >= 2);

        if (isExpenseAmt || isExpenseInit || isDpsAmt || isCashAmt) {
            cleanCurrencyInput(e.range);
        }

        // Apply row colors if Status (Col 4) or Summary Table (Col 8, 9) is edited in DPS Tracker
        if (sheetName === "DPS Tracker" && ((startCol <= 4 && endCol >= 4) || (startCol <= 9 && endCol >= 8))) {
            applyDPSRowColors(sheet);
        }
        return;
    }

    // 2. Handlers for Investment Sheet
    if (sheetName === "Investment") {
        const startRow = e.range.getRow();
        const endRow = startRow + e.range.getNumRows() - 1;
        const startCol = e.range.getColumn();
        const endCol = startCol + e.range.getNumColumns() - 1;

        if (endRow >= 3 && startRow <= 50) {
            // Update row colors when Investment Type (Col 2) or Status (Col 7) is edited
            if ((startCol <= 2 && endCol >= 2) || (startCol <= 7 && endCol >= 7)) {
                applyInvestmentRowColors(sheet);
            }

            // Clean currency strings if entered in Amount column (Col 3)
            if (startCol <= 3 && endCol >= 3) {
                const amtRange = sheet.getRange(Math.max(3, startRow), 3, Math.min(50, endRow) - Math.max(3, startRow) + 1, 1);
                cleanCurrencyInput(amtRange, '#,##0"/="');
            }
        }
        return;
    }

    // 3. Handlers for Financial Year Sheets
    if (FY_REGEX.test(sheetName)) {
        const colIndex = e.range.getColumn();
        // Clean user input if currency strings with '/=' or commas are pasted (except Remarks column 9)
        if (colIndex !== 9) {
            cleanCurrencyInput(e.range);
        }
    }
}

function updateAllSheetsManual() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    updateAllSheets(ss.getActiveSheet(), ss);
    SpreadsheetApp.getUi().alert(
        'Update Complete',
        'Master Year and Dashboard updated successfully!\nAll blank rows and columns have also been cleaned.',
        SpreadsheetApp.getUi().ButtonSet.OK
    );
}

function updateAllSheets(targetSheet, ss = null) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    const originalActiveSheet = ss.getActiveSheet();

    const masterSubTotalRow = consolidateMasterYear(ss);
    consolidateInvestmentSheet(ss);
    setupExpensesSheet(ss);
    setupDPSTrackerSheet(ss);
    setupCashInHandSheet(ss);
    consolidateDashboard(targetSheet, masterSubTotalRow, ss);
    trimAllSheets(ss);

    if (originalActiveSheet) {
        ss.setActiveSheet(originalActiveSheet);
    }
}

function applyDPSRowColors(dpsSheet) {
    const maxRows = Math.max(dpsSheet.getMaxRows(), 3);
    const numRows = maxRows - 2;
    if (numRows > 0) {
        const dataValues = dpsSheet.getRange(3, 4, numRows, 1).getValues();
        const bgs = [];

        for (let r = 0; r < dataValues.length; r++) {
            const status = dataValues[r][0];
            let color = "#FFFFFF";

            if (status === "Paid") {
                color = "#E2EFDA"; // Soft Green
            } else if (status === "Pending") {
                color = "#DDEBF7"; // Soft Blue
            }

            bgs.push(Array(5).fill(color));
        }

        dpsSheet.getRange(3, 1, numRows, 5).setBackgrounds(bgs);
    }

    // Summary table colors
    const summaryData = dpsSheet.getRange("H3:I5").getValues();
    const sumBgs = [];
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    for (let r = 0; r < summaryData.length; r++) {
        const expRaw = summaryData[r][0];
        const status = summaryData[r][1];
        const expDate = expRaw ? new Date(expRaw) : null;
        let color = "#FFFFFF";

        if (status === "Close") {
            color = "#FCE4D6"; // Closed (Soft Peach)
        } else if (status === "Active" && expDate && !isNaN(expDate.getTime()) && expDate < today) {
            color = "#FADBD8"; // Warning: Expired Active (Soft Red)
        }

        sumBgs.push(Array(4).fill(color));
    }

    dpsSheet.getRange("G3:J5").setBackgrounds(sumBgs);
}

function consolidateMasterYear(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    let masterSheet = ss.getSheetByName("Master Year");

    if (!masterSheet) {
        masterSheet = ss.insertSheet("Master Year", 1);
    } else {
        masterSheet.clear();
        masterSheet.getRange(1, 1, masterSheet.getMaxRows(), masterSheet.getMaxColumns()).breakApart();
    }

    setupTableHeaders(masterSheet);

    const sheets = ss.getSheets();
    const fySheets = sheets.filter(s => FY_REGEX.test(s.getName()));

    // Sort chronologically by start year
    fySheets.sort((a, b) => parseInt(a.getName().split("-")[0], 10) - parseInt(b.getName().split("-")[0], 10));

    const neededRows = Math.max(4 + (fySheets.length * 13) + 2, 20);
    ensureSheetDimensions(masterSheet, neededRows, 11);

    let currentRow = 4;
    const totalRowsList = [];

    for (let i = 0; i < fySheets.length; i++) {
        const sheet = fySheets[i];
        const sheetName = sheet.getName();
        const startRow = currentRow;

        // Dynamic formulas linking directly to the financial year sheet
        const formulaGrid = [];
        for (let r = 0; r < 12; r++) {
            const srcRow = 4 + r;
            const mRow = startRow + r;
            formulaGrid.push([
                `='${sheetName}'!A${srcRow}`,
                `='${sheetName}'!B${srcRow}`,
                `='${sheetName}'!C${srcRow}`,
                `='${sheetName}'!D${srcRow}`,
                `='${sheetName}'!E${srcRow}`,
                `=SUM(B${mRow}:E${mRow})`,
                `='${sheetName}'!G${srcRow}`,
                `=F${mRow}-N(G${mRow})-N(J${mRow})`,
                `='${sheetName}'!I${srcRow}`,
                `='${sheetName}'!J${srcRow}`,
                `='${sheetName}'!K${srcRow}`
            ]);
        }

        masterSheet.getRange(startRow, 1, 12, 11).setFormulas(formulaGrid);
        masterSheet.getRange(`A${startRow}:A${startRow + 11}`).setNumberFormat("@");
        masterSheet.getRange(`I${startRow}:I${startRow + 11}`).setNumberFormat("@");

        // Alternating backgrounds
        const bgs = [];
        for (let r = startRow; r < startRow + 12; r++) {
            const bg = (r % 2 === 0) ? "#FFFFFF" : "#F8F9FA";
            bgs.push(Array(11).fill(bg));
        }
        masterSheet.getRange(`A${startRow}:K${startRow + 11}`).setBackgrounds(bgs);

        currentRow += 12;

        masterSheet.getRange(`A${currentRow}`).setValue(`Total (${sheetName})`);
        const endRow = currentRow - 1;
        const sumFormulas = [
            `=SUM(B${startRow}:B${endRow})`,
            `=SUM(C${startRow}:C${endRow})`,
            `=SUM(D${startRow}:D${endRow})`,
            `=SUM(E${startRow}:E${endRow})`,
            `=SUM(F${startRow}:F${endRow})`,
            `=SUM(G${startRow}:G${endRow})`,
            `=SUM(H${startRow}:H${endRow})`,
            "",
            `=SUM(J${startRow}:J${endRow})`,
            `=SUM(K${startRow}:K${endRow})`
        ];
        masterSheet.getRange(`B${currentRow}:K${currentRow}`).setFormulas([sumFormulas]);
        masterSheet.getRange(`A${currentRow}:K${currentRow}`).setFontWeight("bold").setBackground("#FCF3CF");
        totalRowsList.push(currentRow);
        currentRow += 1;
    }

    if (totalRowsList.length > 0) {
        masterSheet.getRange(`A${currentRow}`).setValue("SUB TOTAL");

        const cols = ["B", "C", "D", "E", "F", "G", "H", "I", "J", "K"];
        const subTotalFormulas = cols.map(colLetter => {
            if (colLetter === "I") return "";
            return `=SUM(${totalRowsList.map(r => `${colLetter}${r}`).join(",")})`;
        });
        masterSheet.getRange(`B${currentRow}:K${currentRow}`).setFormulas([subTotalFormulas]);

        masterSheet.getRange(`A${currentRow}:K${currentRow}`)
            .setFontWeight("bold")
            .setFontColor("#1B4F72")
            .setBackground("#D4E6F1");

        masterSheet.getRange(`B4:H${currentRow}`).setNumberFormat('#,##0"/="');
        masterSheet.getRange(`I4:I${currentRow}`).setNumberFormat("@");
        masterSheet.getRange(`J4:K${currentRow}`).setNumberFormat('#,##0"/="');

        applyProfessionalBorders(masterSheet.getRange(`A2:K${currentRow}`));
    }

    masterSheet.autoResizeColumns(1, 11);
    trimSheet(masterSheet, currentRow, 11);
    return currentRow;
}

function consolidateInvestmentSheet(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    let invSheet = ss.getSheetByName("Investment");

    if (!invSheet) {
        invSheet = ss.insertSheet("Investment");
    }

    ensureSheetDimensions(invSheet, 50, 8);

    invSheet.getRange("A2:H2").breakApart();

    invSheet.getRange("A1:H1").merge()
        .setValue("Investment Details")
        .setFontWeight("bold")
        .setFontSize(14)
        .setFontColor("#FFFFFF")
        .setBackground("#1B4F72")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    invSheet.getRange("A2:H2").setValues([["#", "Investment Type", "Amount", "Issue date", "Expired date", "Duration Year", "Active Status", "Institution / Remarks"]]);
    invSheet.getRange("A2:H2").setFontWeight("bold").setBackground("#EAEDED").setHorizontalAlignment("center").setVerticalAlignment("middle");

    // Clean and convert any text-formatted currency strings in Column C
    const amtRange = invSheet.getRange("C3:C50");
    cleanCurrencyInput(amtRange);

    // Clean whitespace from Type (Col B) and Status (Col G)
    const bgRange = invSheet.getRange("B3:G50");
    const bgVals = bgRange.getValues();
    let hasCleanedBG = false;
    for (let i = 0; i < bgVals.length; i++) {
        const typeStr = bgVals[i][0];
        const statusStr = bgVals[i][5];
        if (typeof typeStr === 'string' && typeStr !== typeStr.trim()) {
            bgVals[i][0] = typeStr.trim();
            hasCleanedBG = true;
        }
        if (typeof statusStr === 'string' && statusStr !== statusStr.trim()) {
            bgVals[i][5] = statusStr.trim();
            hasCleanedBG = true;
        }
    }
    if (hasCleanedBG) {
        bgRange.setValues(bgVals);
    }

    const ruleType = SpreadsheetApp.newDataValidation()
        .requireValueInList(["Shonchoypotro", "Business", "In Savings Account"], true)
        .setAllowInvalid(false)
        .build();
    invSheet.getRange("B3:B50").setDataValidation(ruleType);

    const ruleStatus = SpreadsheetApp.newDataValidation()
        .requireValueInList(["Active", "Close"], true)
        .setAllowInvalid(false)
        .build();
    invSheet.getRange("G3:G50").setDataValidation(ruleStatus);

    // Ensure Column H (Institution / Remarks) is free-text and clears any leaked dropdowns
    invSheet.getRange("H3:H50").clearDataValidations();

    invSheet.getRange("A3:A50").setNumberFormat("0");
    invSheet.getRange("C3:C50").setNumberFormat('#,##0"/="');
    invSheet.getRange("D3:E50").setNumberFormat("yyyy-mm-dd");
    invSheet.getRange("F3:F50").setNumberFormat("0");

    applyProfessionalBorders(invSheet.getRange("A1:H50"));
    applyInvestmentRowColors(invSheet);
    invSheet.autoResizeColumns(1, 8);
    trimSheet(invSheet, 50, 8);
}

function applyInvestmentRowColors(invSheet) {
    // Single batched API call for performance (reads B3 to G50)
    const dataValues = invSheet.getRange("B3:G50").getValues();
    const bgs = [];

    const today = new Date();
    today.setHours(0, 0, 0, 0);

    for (let r = 0; r < dataValues.length; r++) {
        const type = dataValues[r][0];     // Col B (Index 0)
        const expRaw = dataValues[r][3];   // Col E (Index 3)
        const status = dataValues[r][5];   // Col G (Index 5)
        const expDate = expRaw ? new Date(expRaw) : null;
        let color = "#FFFFFF";

        if (status === "Close") {
            color = "#FCE4D6"; // Closed (Soft Peach)
        } else if (status === "Active" && expDate && !isNaN(expDate.getTime()) && expDate < today) {
            color = "#FADBD8"; // Warning: Matured / Expired Active Investment (Soft Red)
        } else if (type === "Shonchoypotro") {
            color = "#E2EFDA"; // Soft Green
        } else if (type === "Business") {
            color = "#FFF2CC"; // Soft Yellow
        } else if (type === "In Savings Account") {
            color = "#EDE7F6"; // Soft Lavender / Purple
        }

        bgs.push(Array(8).fill(color)); // Fill A to H with the selected background
    }

    invSheet.getRange("A3:H50").setBackgrounds(bgs);
}

function consolidateDashboard(targetSheet, masterSubTotalRow, ss) {

        dashboard = ss.insertSheet("Dashboard", 0);
    }

    const sheets = ss.getSheets();
    const fySheets = sheets.filter(s => FY_REGEX.test(s.getName()));
    fySheets.sort((a, b) => parseInt(a.getName().split("-")[0], 10) - parseInt(b.getName().split("-")[0], 10));

    if (!masterSubTotalRow) {
        masterSubTotalRow = 4 + (fySheets.length * 13);
    }

    let activeFYSheet = targetSheet;
    if (!activeFYSheet || !FY_REGEX.test(activeFYSheet.getName())) {
        activeFYSheet = fySheets.length > 0 ? fySheets[fySheets.length - 1] : null;
    }
    const activeFYName = activeFYSheet ? activeFYSheet.getName() : "";

    // Precalculate exact layout bounds to avoid extra blank pages in PDF export
    const dropdownRow = 15;
    const activeTableStart = dropdownRow + 2;
    const dropdownCellRef = `$C$${dropdownRow}`;

    const yearlyStartRow = Math.max(activeTableEnd + 3, 32);

function createModernScorecard(sheet, rangeStr, title, value, accentColor, customFormat) {
    const range = sheet.getRange(rangeStr);
    range.setBackground("#FFFFFF");
    range.setBorder(true, true, true, true, false, false, "#DADCE0", SpreadsheetApp.BorderStyle.SOLID);

    const startRow = range.getRow();
    const startCol = range.getColumn();
    const numCols = range.getNumColumns();

    // Top accent bar
    const topRow = sheet.getRange(startRow, startCol, 1, numCols);
    topRow.setBorder(true, true, false, true, false, false, accentColor, SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    // Break apart to avoid overlapping merge exceptions from shifting layouts
    sheet.getRange(startRow + 1, startCol, 2, numCols).breakApart();

    // Title row (merged across card columns)
    sheet.getRange(startRow + 1, startCol, 1, numCols).merge()
        .setValue(title)
        .setFontColor("#5F6368")
        .setFontSize(9)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // Value row (merged across card columns)
    const valRange = sheet.getRange(startRow + 2, startCol, 1, numCols).merge();
    if (typeof value === 'string' && value.startsWith('=')) {
        valRange.setFormula(value);
    } else {
        valRange.setValue(value);
    }
    valRange.setFontColor("#202124")
        .setFontSize(16)
        .setFontWeight("bold")
        .setNumberFormat(customFormat || '#,##0"/="')
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");
}

function trimAllSheets(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheets = ss.getSheets();

    sheets.forEach(sheet => {
        const sName = sheet.getName();

        if (FY_REGEX.test(sName)) {
            trimSheet(sheet, 16, 11);
        } else if (sName === "Investment") {
            // FIX: Trimming to 8 columns instead of 7 to preserve the Remarks column
            trimSheet(sheet, 50, 8);
        } else if (sName === "Expenses") {
            trimSheet(sheet, Math.max(sheet.getLastRow(), 50), 5);
        } else if (sName === "DPS Tracker") {
            trimSheet(sheet, Math.max(sheet.getLastRow(), 50), 10);
        } else if (sName === "Cash in Hand") {
            trimSheet(sheet, 6, 3);
        } else if (sName === "Master Year") {
            const lastRow = sheet.getLastRow();
            if (lastRow > 0) {
                trimSheet(sheet, Math.max(lastRow, 16), 11);
            }
        } else if (sName === "Dashboard") {
            trimSheet(sheet, getDashboardTargetRows(ss), 13);
        }
    });
}

function getDashboardTargetRows(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    const dashboard = ss.getSheetByName("Dashboard");
    const sheets = ss.getSheets();

    let fySheetsCount = 0;
    sheets.forEach(s => {
        if (FY_REGEX.test(s.getName())) fySheetsCount++;
    });

    const pfDataCount = fySheetsCount > 1 ? fySheetsCount : 12;
    const dropdownRow = 15;
    const activeTableStart = dropdownRow + 2;
    const yearlyStartRow = Math.max(activeTableStart + 12 + 3, 32);
    const grandTotalRow = yearlyStartRow + (fySheetsCount > 0 ? fySheetsCount : 1) + 1;
    const invTableStart = Math.max(grandTotalRow + 3, 42);
    const invTotalRow = invTableStart + 8;
    const pfStartRow = Math.max(invTotalRow + 3, 52);
    const pfTotalRow = pfStartRow + 1 + pfDataCount;

    const chart1Row = 15;
    const chart2Row = Math.max(chart1Row + 16, activeTableStart);
    const chart3Row = Math.max(chart2Row + 16, yearlyStartRow);
    const chart4Row = Math.max(chart3Row + 16, invTableStart);

    // Keep 3 blank rows below the last graph so it is never cut off
    let calculatedLastRow = Math.max(chart4Row + 16, pfTotalRow + 2) + 3;

    if (dashboard) {
        const charts = dashboard.getCharts();
        charts.forEach(chart => {
            try {
                const container = chart.getContainerInfo();
                if (container) {
                    const anchorRow = container.getAnchorRow();
                    const chartBottomWithBuffer = anchorRow + 16 + 3;
                    if (chartBottomWithBuffer > calculatedLastRow) {
                        calculatedLastRow = chartBottomWithBuffer;
                    }
                }
            } catch (e) { }
        });
    }

    return calculatedLastRow;
}

function ensureSheetDimensions(sheet, minRows, minCols) {
    if (!sheet) return;
    const currentRows = sheet.getMaxRows();
    if (currentRows < minRows) {
        sheet.insertRowsAfter(currentRows, minRows - currentRows);
    }
    const currentCols = sheet.getMaxColumns();
    if (currentCols < minCols) {
        sheet.insertColumnsAfter(currentCols, minCols - currentCols);
    }
}

function trimSheet(sheet, targetRows, targetCols) {
    if (!sheet) return;
    const currentRows = sheet.getMaxRows();
    if (currentRows > targetRows) {
        sheet.deleteRows(targetRows + 1, currentRows - targetRows);
    }
    const currentCols = sheet.getMaxColumns();
    if (currentCols > targetCols) {
        sheet.deleteColumns(targetCols + 1, currentCols - targetCols);
    }
}

function generatePdfServer(sheetId, fitVal) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const ssId = ss.getId();
    
    // FORCE DRIVE SCOPE: This dummy call ensures Google Apps Script requests 
    // the correct Google Drive permissions for ScriptApp.getOAuthToken().
    DriveApp.getFileById(ssId);
    
    let url = 'https://docs.google.com/spreadsheets/d/' + ssId + '/export?format=pdf&size=7&portrait=false&gridlines=false&printtitle=false&fzr=false&top_margin=0.25&bottom_margin=0.25&left_margin=0.25&right_margin=0.25&scale=' + fitVal;

    let filename = ss.getName();
    if (sheetId === 'ALL') {
        url += '&sheetnames=true&pagenumbers=true';
        filename += ' - Full Report.pdf';
    } else {
        url += '&sheetnames=false&pagenumbers=false&gid=' + sheetId;
        const sheet = getSheetById(ss, sheetId);
        if (sheet) {
            filename += ' - ' + sheet.getName() + '.pdf';
        } else {
            filename += ' - Export.pdf';
        }
    }

    const token = ScriptApp.getOAuthToken();
    const response = UrlFetchApp.fetch(url, {
        headers: { 'Authorization': 'Bearer ' + token },
        muteHttpExceptions: true
    });

    if (response.getResponseCode() !== 200) {
        throw new Error("Unable to export PDF. Code: " + response.getResponseCode());
    }

    const blob = response.getBlob();
    return {
        data: Utilities.base64Encode(blob.getBytes()),
        filename: filename
    };
}

function getSheetById(ss, id) {
    const sheets = ss.getSheets();
    for (let i = 0; i < sheets.length; i++) {
        if (sheets[i].getSheetId().toString() === id.toString()) {
            return sheets[i];
        }
    }
    return null;
}
