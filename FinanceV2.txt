function cardValueRef(rangeStr) {
    const m = rangeStr.match(/^([A-Z]+)(\d+):/);
    return `${m[1]}${parseInt(m[2], 10) + 2}`;
}

function cv(key) {
    return cardValueRef(CARDS[key].range);
}

function parseFY(text) {
    const m = String(text).match(/^(\d{4})-(\d{2}|\d{4})$/);
    if (!m) return null;
    const start = parseInt(m[1], 10);
    const end = m[2].length === 2
        ? Math.floor(start / 100) * 100 + parseInt(m[2], 10)
        : parseInt(m[2], 10);
    return { start: start, end: end };
}

function getFYSheetsSorted(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    return ss.getSheets()
        .filter(s => FY_REGEX.test(s.getName()))
        .sort((a, b) => parseInt(a.getName().split("-")[0], 10) - parseInt(b.getName().split("-")[0], 10));
}

function addMonths(date, n) {
    const d = new Date(date.getFullYear(), date.getMonth() + n, 1);
    const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
    d.setDate(Math.min(date.getDate(), lastDay));
    return d;
}

function parseCurrency(val) {
    if (typeof val === "number") return val;
    if (typeof val !== "string") return NaN;
    const m = val.trim().match(/^(?:৳|\$|bdt|tk\.?)?\s*(-?\d[\d,]*(?:\.\d+)?)\s*(?:\/=|\/-|৳|bdt|tk\.?|taka)?$/i);
    return m ? parseFloat(m[1].replace(/,/g, "")) : NaN;
}

function tz() {
    return Session.getScriptTimeZone();
}

function getOrCreateFolder(name) {
    const it = DriveApp.getFoldersByName(name);
    return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}

function ensureSheetDimensions(sheet, minRows, minCols) {
    if (!sheet) return;
    const currentRows = sheet.getMaxRows();
    if (currentRows < minRows) sheet.insertRowsAfter(currentRows, minRows - currentRows);
    const currentCols = sheet.getMaxColumns();
    if (currentCols < minCols) sheet.insertColumnsAfter(currentCols, minCols - currentCols);
}

function trimSheet(sheet, targetRows, targetCols) {
    if (!sheet) return;
    // Never try to delete every non-frozen row/column
    targetRows = Math.max(targetRows, sheet.getFrozenRows() + 1, 1);
    targetCols = Math.max(targetCols, sheet.getFrozenColumns() + 1, 1);
    const currentRows = sheet.getMaxRows();
    if (currentRows > targetRows) sheet.deleteRows(targetRows + 1, currentRows - targetRows);
    const currentCols = sheet.getMaxColumns();
    if (currentCols > targetCols) sheet.deleteColumns(targetCols + 1, currentCols - targetCols);
}

function applyProfessionalBorders(range) {
    range.setBorder(true, true, true, true, true, true, CONFIG.COLORS.border, SpreadsheetApp.BorderStyle.SOLID);
}

function setPercentRules(sheet, a1) {
    const target = sheet.getRange(a1);
    const kept = sheet.getConditionalFormatRules().filter(r => {
        const rs = r.getRanges();
        return !(rs.length === 1 && rs[0].getA1Notation() === target.getA1Notation());
    });
    const over = SpreadsheetApp.newConditionalFormatRule()
        .whenNumberGreaterThanOrEqualTo(1).setBackground("#F4C7C3").setFontColor("#B31412").setRanges([target]).build();
    const warn = SpreadsheetApp.newConditionalFormatRule()
        .whenNumberBetween(0.8, 0.99999).setBackground("#FCE8B2").setFontColor("#B06000").setRanges([target]).build();
    sheet.setConditionalFormatRules(kept.concat([over, warn]));
}

function addNewYearPrompt() {
    const ui = SpreadsheetApp.getUi();
    const result = ui.prompt(
        'Add New Financial Year Sheet',
        'Enter the financial year range (e.g., 2025-2026 or 2025-26):',
        ui.ButtonSet.OK_CANCEL
    );
    if (result.getSelectedButton() !== ui.Button.OK) return;

    const parsed = parseFY(result.getResponseText().trim());
    if (!parsed) {
        ui.alert('Invalid Format', 'Please enter a valid financial year range like "2025-2026" or "2025-26".', ui.ButtonSet.OK);
        return;
    }
    if (parsed.end !== parsed.start + 1) {
        ui.alert('Invalid Year Range', 'The second year must be exactly one year after the first year (e.g., 2025-2026).', ui.ButtonSet.OK);
        return;
    }

    const fyName = `${parsed.start}-${parsed.end}`;
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // Offer to copy figures from the closest earlier year
    let copyFrom = null;
    const earlier = getFYSheetsSorted(ss).filter(s => parseInt(s.getName().split("-")[0], 10) < parsed.start);
    if (earlier.length > 0) {
        const prev = earlier[earlier.length - 1];
        const ans = ui.alert(
            'Copy previous year?',
            `Pre-fill Basic Salary, House Rent, Conveyance, Medical, Tax and Provident Fund from ${prev.getName()}?\n(Savings and Remarks are not copied.)`,
            ui.ButtonSet.YES_NO
        );
        if (ans === ui.Button.YES) copyFrom = prev.getName();
    }

    addNewYearSheet(fyName, parsed.start, parsed.end, copyFrom);
}

function addNewYearSheet(fyRange, startYear, endYear, copyFromName) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (ss.getSheetByName(fyRange)) {
        SpreadsheetApp.getUi().alert(`A sheet named "${fyRange}" already exists!`);
        return;
    }

    if (!startYear || !endYear) {
        const p = parseFY(fyRange);
        startYear = p.start;
        endYear = p.end;
    }

    const newSheet = ss.insertSheet(fyRange);
    setupTableHeaders(newSheet);

    const months = [
        [`July, ${startYear}`], [`August, ${startYear}`], [`September, ${startYear}`],
        [`October, ${startYear}`], [`November, ${startYear}`], [`December, ${startYear}`],
        [`January, ${endYear}`], [`February, ${endYear}`], [`March, ${endYear}`],
        [`April, ${endYear}`], [`May, ${endYear}`], [`June, ${endYear}`]
    ];
    newSheet.getRange("A4:A15").setValues(months).setNumberFormat("@");

    newSheet.getRange("F4:F15").setFormula("=SUM(B4:E4)");
    newSheet.getRange("H4:H15").setFormula("=F4-N(G4)-N(J4)");

    // Optional copy of the previous year's inputs
    if (copyFromName) {
        const prev = ss.getSheetByName(copyFromName);
        if (prev) {
            newSheet.getRange("B4:E15").setValues(prev.getRange("B4:E15").getValues());
            newSheet.getRange("G4:G15").setValues(prev.getRange("G4:G15").getValues());
            newSheet.getRange("J4:J15").setValues(prev.getRange("J4:J15").getValues());
        }
    }

    const bgs = [];
    for (let r = 4; r <= 15; r++) {
        bgs.push(Array(11).fill(r % 2 === 0 ? CONFIG.COLORS.white : CONFIG.COLORS.rowAlt));
    }
    newSheet.getRange("A4:K15").setBackgrounds(bgs);

    newSheet.getRange("A16").setValue(`Total (${fyRange})`);
    const sumFormulas = [
        "=SUM(B4:B15)", "=SUM(C4:C15)", "=SUM(D4:D15)", "=SUM(E4:E15)",
        "=SUM(F4:F15)", "=SUM(G4:G15)", "=SUM(H4:H15)", "",
        "=SUM(J4:J15)", "=SUM(K4:K15)"
    ];
    newSheet.getRange("B16:K16").setFormulas([sumFormulas]);
    newSheet.getRange("A16:K16").setFontWeight("bold").setBackground(CONFIG.COLORS.fyTotal);

    newSheet.getRange("B4:H16").setNumberFormat(CURRENCY_FMT);
    newSheet.getRange("I4:I16").setNumberFormat("@");
    newSheet.getRange("J4:K16").setNumberFormat(CURRENCY_FMT);

    applyProfessionalBorders(newSheet.getRange("A2:K16"));
    newSheet.autoResizeColumns(1, 11);
    trimSheet(newSheet, 16, 11);

    updateAllSheets(newSheet, ss, true);
    SpreadsheetApp.getUi().alert('Success', `Sheet for Financial Year ${fyRange} created!`, SpreadsheetApp.getUi().ButtonSet.OK);
}

function setupTableHeaders(sheet) {
    sheet.getRange("A1:K3").breakApart();

    sheet.getRange("A2:A3").merge().setValue("Month/Year");
    sheet.getRange("I2:I3").merge().setValue("Remarks");
    sheet.getRange("J2:J3").merge().setValue("Provident Fund");
    sheet.getRange("K2:K3").merge().setValue("Monthly Savings");
    sheet.getRange("B2:H2").merge().setValue("Salary and Others");

    sheet.getRange("B3:H3").setValues([[
        "Basic Salary", "House Rent", "Conveyance Allowance",
        "Medical & Others", "Total Salary", "Tax", "Net Income"
    ]]);

    sheet.getRange("A2:K3")
        .setFontWeight("bold")
        .setFontColor("#FFFFFF")

        .setFontColor(CONFIG.COLORS.subHeaderText)
        .setBackground(CONFIG.COLORS.subHeader);
}

/**
 * Converts strings such as "12,500", "12,500/=", "৳12500", "Tk 500" into numbers.
 * Anything else (text, month names, remarks) is left untouched.
 */
function cleanCurrencyInput(range, customFormat) {
    const values = range.getValues();
    let hasChanges = false;

    for (let i = 0; i < values.length; i++) {
        for (let j = 0; j < values[i].length; j++) {
            const val = values[i][j];
            if (typeof val === 'string' && val.trim().length > 0) {
                const n = parseCurrency(val);
                if (!isNaN(n)) {
                    values[i][j] = n;
                    hasChanges = true;
                }
            }
        }
    }

    if (hasChanges) {
        range.setValues(values);
        if (customFormat) range.setNumberFormat(customFormat);
    }
}

function logAudit(e, sheet) {
    if (!CONFIG.AUDIT_ENABLED) return;
    const name = sheet.getName();
    if ([SHEETS.DASH, SHEETS.MASTER, SHEETS.AUDIT, SHEETS.NW_HIST].indexOf(name) !== -1) return;

    const ss = e.source || SpreadsheetApp.getActiveSpreadsheet();
    const log = ss.getSheetByName(SHEETS.AUDIT);
    if (!log) return; // created during the first refresh

    const safe = v => (typeof v === 'string' && /^[=+\-@]/.test(v)) ? "'" + v : v;
    const single = e.range.getNumRows() === 1 && e.range.getNumColumns() === 1;
    log.appendRow([
        new Date(), name, e.range.getA1Notation(),
        single ? safe(e.oldValue === undefined ? "" : e.oldValue) : "(multi-cell edit)",
        single ? safe(e.value === undefined ? "" : e.value) : ""
    ]);
    if (log.getLastRow() > CONFIG.AUDIT_MAX_ROWS + 500) log.deleteRows(2, 500);
}

function autoFillInvestmentDates(sheet, startRow, endRow, startCol, endCol) {
    const r1 = Math.max(3, startRow);
    const r2 = Math.min(50, endRow);
    if (r2 < r1) return;

    const editedD = startCol <= 4 && endCol >= 4;
    const editedE = startCol <= 5 && endCol >= 5;
    const editedF = startCol <= 6 && endCol >= 6;
    if (!(editedD || editedE || editedF)) return;

    const rng = sheet.getRange(r1, 4, r2 - r1 + 1, 3); // D:F
    const vals = rng.getValues();
    let changed = false;

    for (let i = 0; i < vals.length; i++) {
        const issue = vals[i][0];
        const exp = vals[i][1];
        const dur = vals[i][2];
        const okIssue = issue instanceof Date && !isNaN(issue.getTime());
        const okDur = typeof dur === 'number' && dur > 0;
        const okExp = exp instanceof Date && !isNaN(exp.getTime());

        if (okIssue && okDur && (editedD || editedF)) {
            vals[i][1] = addMonths(issue, Math.round(dur * 12));
            changed = true;
        } else if (okIssue && okExp && (dur === "" || dur === null)) {
            vals[i][2] = Math.round(((exp - issue) / (365.25 * 86400000)) * 100) / 100;
            changed = true;
        }
    }
    if (changed) rng.setValues(vals);
}

function onEdit(e) {
    if (!e || !e.range) return;

    const sheet = e.range.getSheet();
    const sheetName = sheet.getName();

    try { logAudit(e, sheet); } catch (err) { /* never block editing */ }

    const startCol = e.range.getColumn();
    const endCol = startCol + e.range.getNumColumns() - 1;
    const startRow = e.range.getRow();
    const endRow = startRow + e.range.getNumRows() - 1;

    // 1. Expenses / DPS Tracker / Cash in Hand
    if (sheetName === SHEETS.EXP || sheetName === SHEETS.DPS || sheetName === SHEETS.CASH) {
        const isExpenseAmt = (sheetName === SHEETS.EXP && startCol <= 3 && endCol >= 3);
        const isExpenseInit = (sheetName === SHEETS.EXP && startCol <= 2 && endCol >= 2 && endRow >= 5 && startRow <= 7);
        const isExpenseBudget = (sheetName === SHEETS.EXP && startCol <= 8 && endCol >= 8 && endRow >= 11);
        const isDpsAmt = (sheetName === SHEETS.DPS && startCol <= 3 && endCol >= 3);
        const isCashAmt = (sheetName === SHEETS.CASH && startCol <= 2 && endCol >= 2);

        if (isExpenseAmt || isExpenseInit || isExpenseBudget || isDpsAmt || isCashAmt) {
            cleanCurrencyInput(e.range);
        }

        if (sheetName === SHEETS.DPS && ((startCol <= 4 && endCol >= 4) || (startCol <= 9 && endCol >= 8))) {
            applyDPSRowColors(sheet);
        }
        return;
    }

    // 2. Investment
    if (sheetName === SHEETS.INV) {
        if (endRow >= 3 && startRow <= 50) {
            if ((startCol <= 2 && endCol >= 2) || (startCol <= 7 && endCol >= 7) || (startCol <= 5 && endCol >= 5)) {
                applyInvestmentRowColors(sheet);
            }
            if (startCol <= 3 && endCol >= 3) {
                const r1 = Math.max(3, startRow);
                const r2 = Math.min(50, endRow);
                cleanCurrencyInput(sheet.getRange(r1, 3, r2 - r1 + 1, 1), CURRENCY_FMT);
            }
            autoFillInvestmentDates(sheet, startRow, endRow, startCol, endCol);
            applyInvestmentRowColors(sheet);
        }
        return;
    }

    // 3. Liabilities
    if (sheetName === SHEETS.LIAB) {
        if (startCol <= 5 && endCol >= 3) cleanCurrencyInput(e.range);
        return;
    }

    // 4. Financial year sheets (Remarks column 9 is free text)
    if (FY_REGEX.test(sheetName)) {
        if (!(startCol === 9 && endCol === 9)) cleanCurrencyInput(e.range);
    }
}

function updateAllSheetsManual() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    updateAllSheets(ss.getActiveSheet(), ss, false);
    SpreadsheetApp.getUi().alert(
        'Update Complete',

function updateAllSheets(targetSheet, ss, forceTarget) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();

    const lock = LockService.getDocumentLock();
    if (!lock.tryLock(30000)) {
        throw new Error("Another update is already running. Please try again in a moment.");
    }

    try {
        const originalActiveSheet = ss.getActiveSheet();

        const masterSubTotalRow = consolidateMasterYear(ss);
        consolidateInvestmentSheet(ss);
        setupExpensesSheet(ss);
        setupDPSTrackerSheet(ss);
        setupCashInHandSheet(ss);
        setupLiabilitiesSheet(ss);
        ensureAuditSheet(ss);
        consolidateDashboard(targetSheet, masterSubTotalRow, ss, forceTarget);
        trimAllSheets(ss);

        if (originalActiveSheet) ss.setActiveSheet(originalActiveSheet);
    } finally {
        lock.releaseLock();
    }
}

function ensureAuditSheet(ss) {
    if (!CONFIG.AUDIT_ENABLED) return;
    let log = ss.getSheetByName(SHEETS.AUDIT);
    if (!log) {
        log = ss.insertSheet(SHEETS.AUDIT);
        log.getRange("A1:E1").setValues([["Time", "Sheet", "Cell", "Old Value", "New Value"]])
            .setFontWeight("bold").setBackground(CONFIG.COLORS.header).setFontColor("#FFFFFF");
        log.setFrozenRows(1);
        log.setColumnWidth(1, 150);
        log.hideSheet();
    }
}

function applyDPSRowColors(dpsSheet) {
    const maxRows = Math.max(dpsSheet.getMaxRows(), 3);
    const numRows = maxRows - 2;
    if (numRows > 0) {
        const dataValues = dpsSheet.getRange(3, 4, numRows, 1).getValues();
        const bgs = dataValues.map(row => {
            let color = CONFIG.COLORS.white;
            if (row[0] === "Paid") color = CONFIG.COLORS.paid;
            else if (row[0] === "Pending") color = CONFIG.COLORS.pending;
            return Array(5).fill(color);
        });
        dpsSheet.getRange(3, 1, numRows, 5).setBackgrounds(bgs);
    }

    const summaryData = dpsSheet.getRange("H3:I5").getValues();
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const sumBgs = summaryData.map(row => {
        const expDate = row[0] ? new Date(row[0]) : null;
        let color = CONFIG.COLORS.white;
        if (row[1] === "Close") color = CONFIG.COLORS.closed;
        else if (row[1] === "Active" && expDate && !isNaN(expDate.getTime()) && expDate < today) color = CONFIG.COLORS.expired;
        return Array(4).fill(color);
    });
    dpsSheet.getRange("G3:J5").setBackgrounds(sumBgs);
}

function generateDpsSchedulePrompt() {
    const ui = SpreadsheetApp.getUi();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    setupDPSTrackerSheet(ss);
    const sh = ss.getSheetByName(SHEETS.DPS);

    const ask = (title, msg) => {
        const r = ui.prompt(title, msg, ui.ButtonSet.OK_CANCEL);
        return r.getSelectedButton() === ui.Button.OK ? r.getResponseText().trim() : null;
    };

    const acctIn = ask('DPS Schedule (1/4)', `DPS account (${CONFIG.DPS_ACCOUNTS.join(", ")}):`);
    if (acctIn === null) return;
    const idx = CONFIG.DPS_ACCOUNTS.findIndex(a => a.toLowerCase() === acctIn.toLowerCase());
    if (idx === -1) { ui.alert('Unknown account', `Use one of: ${CONFIG.DPS_ACCOUNTS.join(", ")}`, ui.ButtonSet.OK); return; }

    const dateIn = ask('DPS Schedule (2/4)', 'First installment date (yyyy-mm-dd):');
    if (dateIn === null) return;
    const dm = dateIn.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (!dm) { ui.alert('Invalid date', 'Use the format yyyy-mm-dd, e.g. 2026-11-05', ui.ButtonSet.OK); return; }
    const start = new Date(parseInt(dm[1], 10), parseInt(dm[2], 10) - 1, parseInt(dm[3], 10));

    const amtIn = ask('DPS Schedule (3/4)', 'Monthly installment amount:');
    if (amtIn === null) return;
    const amount = parseCurrency(amtIn);
    if (isNaN(amount) || amount <= 0) { ui.alert('Invalid amount', 'Enter a positive number.', ui.ButtonSet.OK); return; }

    const tenIn = ask('DPS Schedule (4/4)', 'Tenure in months (e.g., 60):');
    if (tenIn === null) return;
    const tenure = parseInt(tenIn, 10);
    if (isNaN(tenure) || tenure < 1 || tenure > 240) { ui.alert('Invalid tenure', 'Enter a number between 1 and 240.', ui.ButtonSet.OK); return; }

    // First empty row
    const colVals = sh.getRange("A3:C1000").getValues();
    let firstEmpty = colVals.findIndex(r => r[0] === "" && r[1] === "" && r[2] === "");
    if (firstEmpty === -1) { ui.alert('Sheet full', 'No free rows left (limit 1000).', ui.ButtonSet.OK); return; }
    const startRow = 3 + firstEmpty;
    if (startRow + tenure - 1 > 1000) { ui.alert('Not enough rows', 'The schedule does not fit before row 1000.', ui.ButtonSet.OK); return; }

    const account = CONFIG.DPS_ACCOUNTS[idx];
    const rows = [];
    for (let i = 0; i < tenure; i++) rows.push([addMonths(start, i), account, amount, "Pending", ""]);
    sh.getRange(startRow, 1, tenure, 5).setValues(rows);

    // Update the summary row (matured date + active)
    sh.getRange(3 + idx, 8).setValue(addMonths(start, tenure));
    sh.getRange(3 + idx, 9).setValue("Active");

    applyDPSRowColors(sh);
    ui.alert('Schedule created', `${tenure} installments added for ${account} (rows ${startRow}-${startRow + tenure - 1}).\nMatured date set to ${Utilities.formatDate(addMonths(start, tenure), tz(), "yyyy-MM-dd")}.`, ui.ButtonSet.OK);
}

function consolidateMasterYear(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    let masterSheet = ss.getSheetByName(SHEETS.MASTER);

    if (!masterSheet) {
        masterSheet = ss.insertSheet(SHEETS.MASTER, 1);
    } else {
        masterSheet.clear();
        masterSheet.getRange(1, 1, masterSheet.getMaxRows(), masterSheet.getMaxColumns()).breakApart();
    }

    setupTableHeaders(masterSheet);

    const fySheets = getFYSheetsSorted(ss);
    const neededRows = Math.max(4 + (fySheets.length * 13) + 2, 20);
    ensureSheetDimensions(masterSheet, neededRows, 11);

    let currentRow = 4;
    const totalRowsList = [];

    for (let i = 0; i < fySheets.length; i++) {
        const sheetName = fySheets[i].getName();
        const startRow = currentRow;

        // Everything links directly to the FY sheet (single source of truth)
        const formulaGrid = [];
        for (let r = 0; r < 12; r++) {
            const src = 4 + r;
            formulaGrid.push(["A", "B", "C", "D", "E", "F", "G", "H", "I", "J", "K"].map(c => `='${sheetName}'!${c}${src}`));
        }

        masterSheet.getRange(startRow, 1, 12, 11).setFormulas(formulaGrid);
        masterSheet.getRange(`A${startRow}:A${startRow + 11}`).setNumberFormat("@");
        masterSheet.getRange(`I${startRow}:I${startRow + 11}`).setNumberFormat("@");

        const bgs = [];
        for (let r = startRow; r < startRow + 12; r++) {
            bgs.push(Array(11).fill(r % 2 === 0 ? CONFIG.COLORS.white : CONFIG.COLORS.rowAlt));
        }
        masterSheet.getRange(`A${startRow}:K${startRow + 11}`).setBackgrounds(bgs);

        currentRow += 12;

        masterSheet.getRange(`A${currentRow}`).setValue(`Total (${sheetName})`);
        const endRow = currentRow - 1;
        const sumFormulas = ["B", "C", "D", "E", "F", "G", "H", "I", "J", "K"].map(c =>
            c === "I" ? "" : `=SUM(${c}${startRow}:${c}${endRow})`
        );
        masterSheet.getRange(`B${currentRow}:K${currentRow}`).setFormulas([sumFormulas]);
        masterSheet.getRange(`A${currentRow}:K${currentRow}`).setFontWeight("bold").setBackground(CONFIG.COLORS.masterTotal);
        totalRowsList.push(currentRow);
        currentRow += 1;
    }

    if (totalRowsList.length > 0) {
        masterSheet.getRange(`A${currentRow}`).setValue("SUB TOTAL");
        const subTotalFormulas = ["B", "C", "D", "E", "F", "G", "H", "I", "J", "K"].map(c =>
            c === "I" ? "" : `=SUM(${totalRowsList.map(r => `${c}${r}`).join(",")})`
        );
        masterSheet.getRange(`B${currentRow}:K${currentRow}`).setFormulas([subTotalFormulas]);
        masterSheet.getRange(`A${currentRow}:K${currentRow}`)
            .setFontWeight("bold").setFontColor(CONFIG.COLORS.title).setBackground(CONFIG.COLORS.masterSub);

        masterSheet.getRange(`B4:H${currentRow}`).setNumberFormat(CURRENCY_FMT);
        masterSheet.getRange(`I4:I${currentRow}`).setNumberFormat("@");
        masterSheet.getRange(`J4:K${currentRow}`).setNumberFormat(CURRENCY_FMT);

        applyProfessionalBorders(masterSheet.getRange(`A2:K${currentRow}`));
    }

    masterSheet.autoResizeColumns(1, 11);
    trimSheet(masterSheet, currentRow, 11);
    return currentRow;
}

function consolidateInvestmentSheet(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    let invSheet = ss.getSheetByName(SHEETS.INV);
    if (!invSheet) invSheet = ss.insertSheet(SHEETS.INV);

    ensureSheetDimensions(invSheet, 50, 10);

    invSheet.getRange("A1:J2").breakApart();
    invSheet.getRange("A1:J1").merge()
        .setValue("Investment Details")
        .setFontWeight("bold").setFontSize(14).setFontColor("#FFFFFF")
        .setBackground(CONFIG.COLORS.title).setHorizontalAlignment("center").setVerticalAlignment("middle");

    invSheet.getRange("A2:J2").setValues([["#", "Investment Type", "Amount", "Issue date", "Expired date", "Duration Year", "Active Status", "Institution / Remarks", "Interest Rate % (p.a.)", "Est. Maturity (compound)"]]);
    invSheet.getRange("A2:J2").setFontWeight("bold").setBackground(CONFIG.COLORS.subHeader).setHorizontalAlignment("center").setVerticalAlignment("middle").setWrap(true);

    cleanCurrencyInput(invSheet.getRange("C3:C50"));

    // Trim stray whitespace in Type (B) and Status (G)
    const bgRange = invSheet.getRange("B3:G50");
    const bgVals = bgRange.getValues();
    let hasCleaned = false;
    for (let i = 0; i < bgVals.length; i++) {
        [0, 5].forEach(k => {
            if (typeof bgVals[i][k] === 'string' && bgVals[i][k] !== bgVals[i][k].trim()) {
                bgVals[i][k] = bgVals[i][k].trim();
                hasCleaned = true;
            }
        });
    }
    if (hasCleaned) bgRange.setValues(bgVals);

    invSheet.getRange("B3:B50").setDataValidation(
        SpreadsheetApp.newDataValidation().requireValueInList(CONFIG.INV_TYPES, true).setAllowInvalid(false).build()
    );
    invSheet.getRange("G3:G50").setDataValidation(
        SpreadsheetApp.newDataValidation().requireValueInList(["Active", "Close"], true).setAllowInvalid(false).build()
    );
    invSheet.getRange("H3:H50").clearDataValidations();

    // Estimated maturity (compound, annual) - blank unless amount, rate and duration exist
    const matFormulas = [];
    for (let r = 3; r <= 50; r++) {
        matFormulas.push([`=IF(AND(ISNUMBER(C${r}),ISNUMBER(I${r}),ISNUMBER(F${r})),ROUND(C${r}*(1+I${r}/100)^F${r},0),"")`]);
    }
    invSheet.getRange("J3:J50").setFormulas(matFormulas);

    invSheet.getRange("A3:A50").setNumberFormat("0");
    invSheet.getRange("C3:C50").setNumberFormat(CURRENCY_FMT);
    invSheet.getRange("D3:E50").setNumberFormat("yyyy-mm-dd");
    invSheet.getRange("F3:F50").setNumberFormat("General");
    invSheet.getRange("I3:I50").setNumberFormat("0.00");
    invSheet.getRange("J3:J50").setNumberFormat(CURRENCY_FMT);

    applyProfessionalBorders(invSheet.getRange("A1:J50"));
    applyInvestmentRowColors(invSheet);
    invSheet.autoResizeColumns(1, 10);
    trimSheet(invSheet, 50, 10);
}

function applyInvestmentRowColors(invSheet) {
    const dataValues = invSheet.getRange("B3:G50").getValues();
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const bgs = dataValues.map(row => {
        const type = row[0];
        const expRaw = row[3];
        const status = row[5];
        const expDate = expRaw ? new Date(expRaw) : null;
        let color = CONFIG.COLORS.white;

        if (status === "Close") color = CONFIG.COLORS.closed;
        else if (status === "Active" && expDate && !isNaN(expDate.getTime()) && expDate < today) color = CONFIG.COLORS.expired;

        else if (type === CONFIG.INV_TYPES[2]) color = CONFIG.COLORS.savingsAcc;

        return Array(10).fill(color);
    });

    invSheet.getRange("A3:J50").setBackgrounds(bgs);
}

function addExpenseEntry(data) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEETS.EXP);
    if (!sh) throw new Error("Expenses sheet not found. Run 'Update Master Year & Dashboard' first.");

    const amount = parseCurrency(String(data.amount));
    if (isNaN(amount) || amount <= 0) throw new Error("Enter a valid amount.");
    const dm = String(data.date).match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (!dm) throw new Error("Invalid date.");
    if (CONFIG.ACCOUNTS.indexOf(data.paidVia) === -1) throw new Error("Invalid 'Paid Via' value.");

    const date = new Date(parseInt(dm[1], 10), parseInt(dm[2], 10) - 1, parseInt(dm[3], 10));
    const colC = sh.getRange("B11:C1000").getValues();
    const idx = colC.findIndex(r => r[0] === "" && r[1] === "");
    if (idx === -1) throw new Error("No free rows left. Use 'Close Month & Archive'.");

    const row = 11 + idx;
    sh.getRange(row, 1, 1, 5).setValues([[date, data.category, amount, data.paidVia, data.remarks || ""]]);
    sh.getRange(row, 1).setNumberFormat("yyyy-mm-dd");
    return `Saved to row ${row}: ${data.category} - ${amount}`;
}

function closeMonthPrompt() {
    const ui = SpreadsheetApp.getUi();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const exp = ss.getSheetByName(SHEETS.EXP);
    if (!exp) { ui.alert('Expenses sheet not found.'); return; }

    const data = exp.getRange("A11:E1000").getValues().filter(r => r[1] !== "" || r[2] !== "" || r[0] !== "");
    if (data.length === 0) { ui.alert('Nothing to archive', 'The expense list is empty.', ui.ButtonSet.OK); return; }

    const defaultLabel = Utilities.formatDate(new Date(), tz(), "MMMM yyyy");
    const p = ui.prompt('Close Month', `Label for this month's archive (default: ${defaultLabel}):`, ui.ButtonSet.OK_CANCEL);
    if (p.getSelectedButton() !== ui.Button.OK) return;
    const label = p.getResponseText().trim() || defaultLabel;

    const carry = ui.alert('Carry forward balances?',
        "Use each account's Remaining Amount as next month's Initial Amount?\n\nYES = carry forward\nNO = leave Initial Amounts unchanged",
        ui.ButtonSet.YES_NO_CANCEL);
    if (carry === ui.Button.CANCEL || carry === ui.Button.CLOSE) return;

    try { createBackup(true); } catch (err) { /* backup is best-effort */ }

    // Archive
    let hist = ss.getSheetByName(SHEETS.EXP_HIST);
    if (!hist) {
        hist = ss.insertSheet(SHEETS.EXP_HIST);
        hist.getRange("A1:G1").setValues([["Month", "Date", "Category", "Amount", "Paid Via", "Remarks", "Archived On"]])
            .setFontWeight("bold").setBackground(CONFIG.COLORS.header).setFontColor("#FFFFFF").setHorizontalAlignment("center");
        hist.setFrozenRows(1);
        hist.setColumnWidth(1, 120); hist.setColumnWidth(2, 100); hist.setColumnWidth(3, 150);
        hist.setColumnWidth(4, 110); hist.setColumnWidth(5, 110); hist.setColumnWidth(6, 250); hist.setColumnWidth(7, 150);
    }
    const now = new Date();
    const rows = data.map(r => [label, r[0], r[1], r[2], r[3], r[4], now]);
    const startRow = hist.getLastRow() + 1;
    ensureSheetDimensions(hist, startRow + rows.length, 7);
    hist.getRange(startRow, 1, rows.length, 7).setValues(rows);
    hist.getRange(startRow, 2, rows.length, 1).setNumberFormat("yyyy-mm-dd");
    hist.getRange(startRow, 4, rows.length, 1).setNumberFormat(CURRENCY_FMT);
    hist.getRange(startRow, 7, rows.length, 1).setNumberFormat("yyyy-mm-dd hh:mm");

    // Carry forward (read before clearing)
    if (carry === ui.Button.YES) {
        const remaining = exp.getRange("C5:C7").getValues();
        exp.getRange("B5:B7").setValues(remaining.map(r => [typeof r[0] === 'number' ? r[0] : ""]));
    }

    exp.getRange("A11:E1000").clearContent();
    ui.alert('Month closed', `${rows.length} expense rows archived to "${SHEETS.EXP_HIST}" as ${label}.`, ui.ButtonSet.OK);
}

function getDashboardLayout(fyCount) {
    const L = {};
    L.dropdownRow = 20;
    L.activeTableStart = L.dropdownRow + 2;
    L.activeTableEnd = L.activeTableStart + 12;
    L.yearlyStartRow = Math.max(L.activeTableEnd + 3, 37);
    L.grandTotalRow = L.yearlyStartRow + (fyCount > 0 ? fyCount : 1) + 1;
    L.invTableStart = Math.max(L.grandTotalRow + 3, 47);
    L.invTotalRow = L.invTableStart + 8;
    L.pfDataCount = fyCount > 1 ? fyCount : 12;
    L.pfStartRow = Math.max(L.invTotalRow + 3, 57);
    L.pfTotalRow = L.pfStartRow + 1 + L.pfDataCount;

    L.chart1Row = L.dropdownRow;
    L.chart2Row = Math.max(L.chart1Row + 16, L.activeTableStart);
    L.chart3Row = Math.max(L.chart2Row + 16, L.yearlyStartRow);
    L.chart4Row = Math.max(L.chart3Row + 16, L.invTableStart);
    L.chart5Row = L.chart4Row + 16;
    L.lastRow = Math.max(L.chart5Row + 16, L.pfTotalRow + 2) + 3;
    return L;
}

function getDashboardTargetRows(ss) {
    return getDashboardLayout(getFYSheetsSorted(ss).length).lastRow;
}

    // Remember the person's selected year before clearing (also checks the old v1 location C15)
    let savedFY = "";

        [L.dropdownRow, 15].forEach(row => {
            if (!savedFY && row <= dashboard.getMaxRows() && dashboard.getMaxColumns() >= 3) {

function createModernScorecard(sheet, rangeStr, title, value, accentColor, customFormat) {
    const range = sheet.getRange(rangeStr);
    range.setBackground("#FFFFFF");
    range.setBorder(true, true, true, true, false, false, "#DADCE0", SpreadsheetApp.BorderStyle.SOLID);

    const startRow = range.getRow();
    const startCol = range.getColumn();
    const numCols = range.getNumColumns();

    sheet.getRange(startRow, startCol, 1, numCols)
        .setBorder(true, true, false, true, false, false, accentColor, SpreadsheetApp.BorderStyle.SOLID_MEDIUM);

    sheet.getRange(startRow + 1, startCol, 2, numCols).breakApart();

    sheet.getRange(startRow + 1, startCol, 1, numCols).merge()
        .setValue(title).setFontColor("#5F6368").setFontSize(9).setFontWeight("bold")
        .setHorizontalAlignment("center").setVerticalAlignment("middle");

    const valRange = sheet.getRange(startRow + 2, startCol, 1, numCols).merge();
    if (typeof value === 'string' && value.startsWith('=')) valRange.setFormula(value);
    else valRange.setValue(value);
    valRange.setFontColor("#202124").setFontSize(16).setFontWeight("bold")
        .setNumberFormat(customFormat || CURRENCY_FMT)
        .setHorizontalAlignment("center").setVerticalAlignment("middle");
}

function trimAllSheets(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();

    ss.getSheets().forEach(sheet => {
        const n = sheet.getName();
        if (FY_REGEX.test(n)) trimSheet(sheet, 16, 11);
        else if (n === SHEETS.INV) trimSheet(sheet, 50, 10);
        else if (n === SHEETS.EXP) trimSheet(sheet, 1000, 10);   // validation covers rows up to 1000
        else if (n === SHEETS.DPS) trimSheet(sheet, 1000, 10);
        else if (n === SHEETS.CASH) trimSheet(sheet, 6, 3);
        else if (n === SHEETS.LIAB) trimSheet(sheet, 31, 7);
        else if (n === SHEETS.MASTER) {
            const lastRow = sheet.getLastRow();
            if (lastRow > 0) trimSheet(sheet, Math.max(lastRow, 16), 11);
        } else if (n === SHEETS.DASH) trimSheet(sheet, getDashboardTargetRows(ss), 13);
    });
}

function collectAlerts(ss) {
    if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
    const alerts = [];
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const fmt = d => Utilities.formatDate(d, tz(), "yyyy-MM-dd");
    const daysTo = d => Math.ceil((d - today) / 86400000);

    // Investments
    const inv = ss.getSheetByName(SHEETS.INV);
    if (inv) {
        inv.getRange("B3:H50").getValues().forEach(r => {
            const type = r[0], amt = r[1], exp = r[3], status = r[5], note = r[6];
            if (!type || status === "Close" || !(exp instanceof Date)) return;
            const d = daysTo(exp);
            const label = `${type} ${amt !== "" ? "(" + amt + ")" : ""}${note ? " - " + note : ""}`;
            if (d < 0) alerts.push(`MATURED: ${label} expired on ${fmt(exp)} and is still Active.`);
            else if (d <= CONFIG.ALERT_DAYS) alerts.push(`MATURING SOON: ${label} matures in ${d} day(s) (${fmt(exp)}).`);
        });
    }

    // DPS
    const dps = ss.getSheetByName(SHEETS.DPS);
    if (dps) {
        dps.getRange("G3:I5").getValues().forEach(r => {
            if (r[2] !== "Active" || !(r[1] instanceof Date)) return;
            const d = daysTo(r[1]);
            if (d < 0) alerts.push(`DPS MATURED: ${r[0]} matured on ${fmt(r[1])} but is still Active.`);
            else if (d <= CONFIG.ALERT_DAYS) alerts.push(`DPS MATURING: ${r[0]} matures in ${d} day(s) (${fmt(r[1])}).`);
        });

        const pend = dps.getRange("A3:D1000").getValues()
            .filter(r => r[3] === "Pending" && r[0] instanceof Date && r[0] < today);
        if (pend.length > 0) {
            const sample = pend.slice(0, 5).map(r => `${r[1]} ${fmt(r[0])}`).join(", ");
            alerts.push(`DPS OVERDUE: ${pend.length} installment(s) still Pending past their date (e.g. ${sample}).`);
        }
    }

    // Budgets
    const exp = ss.getSheetByName(SHEETS.EXP);
    if (exp) {
        const S = expSummaryRows();
        exp.getRange(S.first, 7, S.last - S.first + 1, 4).getValues().forEach(r => {
            if (typeof r[1] === 'number' && r[1] > 0 && typeof r[2] === 'number') {
                if (r[2] > r[1]) alerts.push(`BUDGET EXCEEDED: ${r[0]} - spent ${r[2]} of ${r[1]}.`);
                else if (r[2] >= r[1] * 0.8) alerts.push(`BUDGET 80%+: ${r[0]} - spent ${r[2]} of ${r[1]}.`);
            }
        });
    }
    return alerts;
}

function showAlertsNow() {
    const alerts = collectAlerts();
    SpreadsheetApp.getUi().alert(
        alerts.length ? `Alerts (${alerts.length})` : 'All clear',
        alerts.length ? alerts.join("\n\n") : 'No maturities, overdue installments or budget issues found.',
        SpreadsheetApp.getUi().ButtonSet.OK
    );
}

function dailyAlertCheck() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (alerts.length === 0) return;
    MailApp.sendEmail(

        `[${ss.getName()}] ${alerts.length} financial alert(s)`,

    );
}

function saveNetWorthSnapshot(silent) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let dash = ss.getSheetByName(SHEETS.DASH);
    if (!dash) {
        updateAllSheets(null, ss, false);
        dash = ss.getSheetByName(SHEETS.DASH);
    }
    SpreadsheetApp.flush();

    const val = key => {
        const v = dash.getRange(cv(key)).getValue();
        return typeof v === 'number' ? v : 0;
    };

    let hist = ss.getSheetByName(SHEETS.NW_HIST);
    if (!hist) {
        hist = ss.insertSheet(SHEETS.NW_HIST);
        hist.getRange("A1:H1").setValues([["Date", "Net Worth", "Active Investments", "Provident Fund", "Grand Total Wealth", "Liabilities", "Net After Debt", "Total Savings"]])
            .setFontWeight("bold").setBackground(CONFIG.COLORS.header).setFontColor("#FFFFFF").setHorizontalAlignment("center");
        hist.setFrozenRows(1);
        hist.setColumnWidth(1, 100);
        for (let c = 2; c <= 8; c++) hist.setColumnWidth(c, 130);
    }

    const now = new Date();
    const row = [now, val("NET_WORTH"), val("ACTIVE_INV"), val("PF"), val("GRAND_TOTAL"), val("LIABILITIES"), val("NET_AFTER_DEBT"), val("SAVINGS")];

    // One snapshot per month: overwrite if this month already exists
    const monthKey = Utilities.formatDate(now, tz(), "yyyy-MM");
    const last = hist.getLastRow();
    let target = last + 1;
    if (last >= 2) {
        const dates = hist.getRange(2, 1, last - 1, 1).getValues();
        for (let i = 0; i < dates.length; i++) {
            if (dates[i][0] instanceof Date && Utilities.formatDate(dates[i][0], tz(), "yyyy-MM") === monthKey) {
                target = 2 + i;
                break;
            }
        }
    }
    ensureSheetDimensions(hist, target, 10);
    hist.getRange(target, 1, 1, 8).setValues([row]);
    hist.getRange(target, 1).setNumberFormat("yyyy-mm-dd");
    hist.getRange(target, 2, 1, 7).setNumberFormat(CURRENCY_FMT);

    rebuildHistoryChart(hist);
    if (!silent) SpreadsheetApp.getUi().alert('Snapshot saved', `Net worth snapshot for ${monthKey} saved to "${SHEETS.NW_HIST}".`, SpreadsheetApp.getUi().ButtonSet.OK);
}

function rebuildHistoryChart(hist) {
    hist.getCharts().forEach(c => hist.removeChart(c));
    const n = hist.getLastRow();
    if (n < 3) return; // need at least 2 data points

    const chart = hist.newChart().asLineChart()

        .addRange(hist.getRange(1, 5, n, 1))   // Grand Total Wealth
        .addRange(hist.getRange(1, 7, n, 1))   // Net After Debt
        .setPosition(2, 10, 0, 0).setNumHeaders(1)
        .setOption('title', 'Wealth Growth Over Time')
        .setOption('legend', { position: 'bottom' })
        .setOption('curveType', 'function').setOption('pointSize', 5)
        .setOption('width', 640).setOption('height', 300)
        .build();
    hist.insertChart(chart);
}

function protectFormulaCells() {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    removeFormulaProtection(true);

    const add = (range, desc) => {
        const p = range.protect().setDescription(`${PROTECT_TAG} ${desc}`);
        p.setWarningOnly(true);
    };

    getFYSheetsSorted(ss).forEach(s => {
        add(s.getRange("F4:F16"), s.getName() + " total salary");
        add(s.getRange("H4:H16"), s.getName() + " net income");
        add(s.getRange("A16:K16"), s.getName() + " totals");
    });

    const master = ss.getSheetByName(SHEETS.MASTER);
    if (master) { const p = master.protect().setDescription(`${PROTECT_TAG} master`); p.setWarningOnly(true); }

    const dash = ss.getSheetByName(SHEETS.DASH);
    if (dash) {
        const p = dash.protect().setDescription(`${PROTECT_TAG} dashboard`);
        p.setUnprotectedRanges([dash.getRange(`C${getDashboardLayout(1).dropdownRow}`)]);
        p.setWarningOnly(true);
    }

    const exp = ss.getSheetByName(SHEETS.EXP);
    if (exp) {
        const S = expSummaryRows();
        add(exp.getRange("B3"), "expenses total salary");
        add(exp.getRange("C5:C7"), "expenses remaining");
        add(exp.getRange(S.first, 9, S.total - S.first + 1, 2), "expenses spent/used");
    }

    const inv = ss.getSheetByName(SHEETS.INV);
    if (inv) add(inv.getRange("J3:J50"), "investment maturity");

    const dps = ss.getSheetByName(SHEETS.DPS);
    if (dps) add(dps.getRange("J3:J5"), "dps totals");

    const cash = ss.getSheetByName(SHEETS.CASH);
    if (cash) add(cash.getRange("B6"), "cash total");

    SpreadsheetApp.getUi().alert('Protection added',
        'Formula cells now show a warning if edited manually (nothing is hard-locked).\nRun "Update Master Year & Dashboard" after adding new sheets and re-run this to cover them.',
        SpreadsheetApp.getUi().ButtonSet.OK);
}

function removeFormulaProtection(silent) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    [SpreadsheetApp.ProtectionType.RANGE, SpreadsheetApp.ProtectionType.SHEET].forEach(type => {
        ss.getProtections(type).forEach(p => {
            if (String(p.getDescription()).indexOf(PROTECT_TAG) === 0) p.remove();
        });
    });
    if (silent !== true) {
        SpreadsheetApp.getUi().alert('Protection removed', 'All formula-cell protections created by this script were removed.', SpreadsheetApp.getUi().ButtonSet.OK);
    }
}

function buildPdf(ss, sheetId, fitVal) {
    const ssId = ss.getId();
    DriveApp.getFileById(ssId); // forces the Drive scope so ScriptApp.getOAuthToken() is valid

    let url = 'https://docs.google.com/spreadsheets/d/' + ssId +
        '/export?format=pdf&size=7&portrait=false&gridlines=false&printtitle=false&fzr=false' +
        '&top_margin=0.25&bottom_margin=0.25&left_margin=0.25&right_margin=0.25&scale=' + fitVal;

    let filename = ss.getName();
    if (sheetId === 'ALL') {
        url += '&sheetnames=true&pagenumbers=true';
        filename += ' - Full Report';
    } else {
        url += '&sheetnames=false&pagenumbers=false&gid=' + sheetId;
        const sheet = getSheetById(ss, sheetId);
        filename += ' - ' + (sheet ? sheet.getName() : 'Export');
    }
    filename += ' (' + Utilities.formatDate(new Date(), tz(), "yyyy-MM-dd") + ').pdf';

    const response = UrlFetchApp.fetch(url, {
        headers: { 'Authorization': 'Bearer ' + ScriptApp.getOAuthToken() },
        muteHttpExceptions: true
    });
    if (response.getResponseCode() !== 200) {
        throw new Error("Unable to export PDF. Code: " + response.getResponseCode());
    }
    return { blob: response.getBlob(), filename: filename };
}

function generatePdfServer(sheetId, fitVal, opts) {
    opts = opts || { download: true };
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const pdf = buildPdf(ss, sheetId, fitVal);
    pdf.blob.setName(pdf.filename);

    const done = [];
    if (opts.email) {
        MailApp.sendEmail({
            to: Session.getEffectiveUser().getEmail(),
            subject: pdf.filename,
            body: "Your exported PDF is attached.",
            attachments: [pdf.blob]
        });
        done.push("emailed");
    }
    if (opts.drive) {
        getOrCreateFolder(CONFIG.REPORT_FOLDER).createFile(pdf.blob);
        done.push("saved to Drive");
    }
    if (opts.download) done.push("downloading");

    return {
        data: opts.download ? Utilities.base64Encode(pdf.blob.getBytes()) : null,
        filename: pdf.filename,
        message: "Done: " + (done.length ? done.join(", ") : "nothing selected") + "."
    };
}

function getSheetById(ss, id) {
    const sheets = ss.getSheets();
    for (let i = 0; i < sheets.length; i++) {
        if (sheets[i].getSheetId().toString() === id.toString()) return sheets[i];
    }
    return null;
}
